-- ============= SLIDER DRAG =============
local flySpeed = 60
local MIN_SPEED = 10
local MAX_SPEED = 300
local sliderDragging = false

local function updateSpeed(value)
	flySpeed = math.clamp(math.floor(value), MIN_SPEED, MAX_SPEED)
	SpeedValue.Text = tostring(flySpeed)
	-- Mise à jour visuelle du fill + knob
	local pct = (flySpeed - MIN_SPEED) / (MAX_SPEED - MIN_SPEED)
	SliderFill.Size = UDim2.new(pct, 0, 1, 0)
	SliderKnob.Position = UDim2.new(pct, -9, 0.5, -9)
end

SliderKnob.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliderDragging = true
		-- Empêcher le drag du bouton fly pendant qu'on drag le slider
		dragging = false
	end
end)

SliderKnob.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliderDragging = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if sliderDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local trackAbsPos = SliderTrack.AbsolutePosition
		local trackAbsSize = SliderTrack.AbsoluteSize
		local relX = (input.Position.X - trackAbsPos.X) / trackAbsSize.X
		relX = math.clamp(relX, 0, 1)
		local newSpeed = MIN_SPEED + (MAX_SPEED - MIN_SPEED) * relX
		updateSpeed(newSpeed)
	end
end)

local function destroyAll()
	-- Stop fly si actif
	if flying then stopFly() end
	-- Cleanup connections
	for _, c in ipairs(_flyConnections) do
		pcall(function() c:Disconnect() end)
	end
	-- Destroy GUI
	pcall(function() FlyGui:Destroy() end)
	print("[FLY LOU] Script désactivé.")
end

-- ============= ROBLOX NATIVE JOYSTICK (PlayerModule) =============
local moveVector = Vector2.new(0, 0)
local joystickActive = false

local function hookRobloxJoystick()
	-- Sur mobile, Roblox PlayerModule fournit un joystick virtuel
	-- On intercepte les inputs de mouvement via UserInputService (Touch + Thumbstick)
	-- et on convertit en Vector2 normalisé pour la direction
	local thumbstickConn = UserInputService:GetPropertyChangedSignal("TouchEnabled"):Connect(function()
		joystickActive = UserInputService.TouchEnabled
	end)
	joystickActive = UserInputService.TouchEnabled
end
hookRobloxJoystick()

-- Listener touch multi-touch : chaque doigt tracke un position
-- 1er doigt = joystick directionnel (calcul par rapport à la position initiale)
-- Roblox utilise son propre ThumbstickFrame pour le mouvement du personnage,
-- mais on peut intercepter les touch events globaux pour le fly
local touchPoints = {}
local joystickTouchId = nil
local joystickStartPos = nil
local joystickCurrentPos = nil
local joystickRadius = 80

UserInputService.TouchStarted:Connect(function(input, gpe)
	if gpe then return end
	if joystickTouchId == nil then
		joystickTouchId = input
		joystickStartPos = input.Position
		joystickCurrentPos = input.Position
	end
end)

UserInputService.TouchMoved:Connect(function(input, gpe)
	if gpe then return end
	if input == joystickTouchId then
		joystickCurrentPos = input.Position
	end
end)

UserInputService.TouchEnded:Connect(function(input, gpe)
	if gpe then return end
	if input == joystickTouchId then
		joystickTouchId = nil
		joystickStartPos = nil
		joystickCurrentPos = nil
		moveVector = Vector2.new(0, 0)
	end
end)

local function getJoystickVector()
	if not joystickStartPos or not joystickCurrentPos then return Vector2.new(0, 0) end
	local delta = joystickCurrentPos - joystickStartPos
	local magnitude = math.min(delta.Magnitude, joystickRadius)
	if delta.Magnitude < 5 then return Vector2.new(0, 0) end
	return Vector2.new(delta.X / joystickRadius, -delta.Y / joystickRadius)
end

local function startFly()
	if flying then return end
	local char = LocalPlayer.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end

	flyBody = Instance.new("BodyVelocity")
	flyBody.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	flyBody.Velocity = Vector3.new(0, 0, 0)
	flyBody.P = 5000
	flyBody.Parent = hrp

	flyGyro = Instance.new("BodyGyro")
	flyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
	flyGyro.D = 100
	flyGyro.P = 10000
	flyGyro.CFrame = Camera.CFrame
	flyGyro.Parent = hrp

	hum.PlatformStand = true
	currentVel = Vector3.new(0, 0, 0)

	flyConn = RunService.RenderStepped:Connect(function()
		if not flying or not hrp or not hrp.Parent then return end
		if flyBody and flyBody.Parent ~= hrp then flyBody.Parent = hrp end
		if flyGyro and flyGyro.Parent ~= hrp then flyGyro.Parent = hrp end

		local camCF = Camera.CFrame
		local lookDir = camCF.LookVector
		local rightDir = camCF.RightVector
		local upDir = Vector3.new(0, 1, 0)

		-- Direction cible (target velocity)
		local targetDir = Vector3.new(0, 0, 0)
		local inputMag = 0

		-- PC : WASD + Space/Shift
		if UserInputService.KeyboardEnabled then
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then targetDir = targetDir + lookDir end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then targetDir = targetDir - lookDir end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then targetDir = targetDir - rightDir end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then targetDir = targetDir + rightDir end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then targetDir = targetDir + upDir end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then targetDir = targetDir - upDir end
			if targetDir.Magnitude > 0 then
				targetDir = targetDir.Unit
				inputMag = 1
			end
		end

		-- Mobile : joystick tactile custom OU Thumbstick Roblox natif
		-- On combine les deux sources pour fluidité max
		if UserInputService.TouchEnabled then
			-- Source 1 : notre joystick custom (touch tracking)
			local customJoy = getJoystickVector()
			if customJoy.Magnitude > 0.05 then
				-- X = strafe (gauche/droite), Y = forward/backward
				-- Note: Y inversé car touch coords Y vers le bas
				targetDir = targetDir + rightDir * customJoy.X + lookDir * customJoy.Y
				inputMag = math.max(inputMag, customJoy.Magnitude)
			end
			-- Source 2 : thumbstick frame de Roblox (PlayerModule)
			-- On lit via le Humanoid MoveDirection si dispo
			if hum and hum.MoveDirection and hum.MoveDirection.Magnitude > 0.05 then
				-- Convertir MoveDirection (world-space) en input normalisé
				local mv = hum.MoveDirection
				-- Projeter sur les axes caméra
				local forwardComponent = mv:Dot(lookDir)
				local rightComponent = mv:Dot(rightDir)
				-- Ajouter seulement si pas déjà couvert par le joystick custom
				if customJoy.Magnitude < 0.1 then
					targetDir = rightDir * rightComponent + lookDir * forwardComponent
					inputMag = math.max(inputMag, mv.Magnitude)
				end
			end
			-- Monter/descendre = inclinaison caméra (sur mobile, le joueur incline l'écran avec 2 doigts)
			-- On utilise la position Y du pouce par rapport à la position centrale
			if joystickStartPos and joystickCurrentPos then
				-- Pas de bouton up/down séparé : la caméra elle-même guide verticalement
				-- Si la caméra pointe vers le haut (lookDir.Y > 0.3) = monte automatiquement
				-- Sinon si elle pointe vers le bas (lookDir.Y < -0.3) = descend
				if customJoy.Magnitude > 0.3 then
					local verticalComponent = lookDir.Y
					if math.abs(verticalComponent) > 0.3 then
						targetDir = targetDir + upDir * verticalComponent
					end
				end
			end
			if targetDir.Magnitude > 0 then
				targetDir = targetDir.Unit
				inputMag = math.max(inputMag, 0.5)
			end
		end

		-- Cible de vélocité
		local targetVel = targetDir * (flySpeed * inputMag)

		-- Lissage fluide (lerp factor basé sur dt pour cohérence framerate)
		-- 0.18 = très fluide, pas saccadé ; 0.3 = plus réactif
		local smoothFactor = 0.18
		currentVel = currentVel:Lerp(targetVel, smoothFactor)

		-- Si input ~0, on coupe net (pas de drift)
		if inputMag < 0.05 then
			currentVel = currentVel * 0.85
			if currentVel.Magnitude < 0.5 then
				currentVel = Vector3.new(0, 0, 0)
			end
		end

		flyBody.Velocity = currentVel
		flyGyro.CFrame = camCF
	end)

	flying = true
	TweenService:Create(ToggleStroke, TweenInfo.new(0.3), {Color = Color3.fromRGB(120, 220, 140)}):Play()
	TweenService:Create(ToggleGlow, TweenInfo.new(0.3), {ImageColor3 = Color3.fromRGB(120, 220, 140), ImageTransparency = 0.4}):Play()
	TweenService:Create(ToggleIcon, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(120, 255, 160)}):Play()
	TweenService:Create(ToggleLabel, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(120, 255, 160)}):Play()
	ToggleLabel.Text = "FLY ON"
end

local function stopFly()
	if not flying then return end
	flying = false
	currentVel = Vector3.new(0, 0, 0)
	if flyConn then flyConn:Disconnect() flyConn = nil end
	if flyBody then flyBody:Destroy() flyBody = nil end
	if flyGyro then flyGyro:Destroy() flyGyro = nil end
	local char = LocalPlayer.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then hum.PlatformStand = false end
	end
	TweenService:Create(ToggleStroke, TweenInfo.new(0.3), {Color = Color3.fromRGB(80, 80, 110)}):Play()
	TweenService:Create(ToggleGlow, TweenInfo.new(0.3), {ImageColor3 = Color3.fromRGB(80, 80, 110), ImageTransparency = 0.7}):Play()
	TweenService:Create(ToggleIcon, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(180, 180, 200)}):Play()
	TweenService:Create(ToggleLabel, TweenInfo.new(0.3), {TextColor3 = Color3.fromRGB(120, 120, 150)}):Play()
	ToggleLabel.Text = "FLY OFF"
end

local function toggleFly()
	if flying then stopFly() else startFly() end
end

ToggleBtn.MouseEnter:Connect(function()
	TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {Size = UDim2.new(0, 76, 0, 76)}):Play()
end)
ToggleBtn.MouseLeave:Connect(function()
	TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {Size = UDim2.new(0, 70, 0, 70)}):Play()
end)

LocalPlayer.Chatted:Connect(function(msg)
	local m = msg:lower()
	if m == ";fly" then
		if not flying then startFly() end
	elseif m == ";unfly" then
		if flying then stopFly() end
	end
end)
